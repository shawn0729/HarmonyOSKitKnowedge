# 如何保存网络图片到相册

---

可以使用安全控件中的保存控件，免去权限申请和权限请求等环节，获得临时授权，保存对应图片。需要申请权限：ohos.permission.INTERNET。参考代码如下：



```cangjie
1. import { http } from '@kit.NetworkKit';
2. import { image } from '@kit.ImageKit';
3. import { BusinessError } from '@kit.BasicServicesKit';
4. import { photoAccessHelper } from '@kit.MediaLibraryKit';
5. import { promptAction } from '@kit.ArkUI';
6. import { fileIo as fs } from '@kit.CoreFileKit';
7.
8. @Entry
9. @Component
10. struct SaveImage {
11.  @State pixelMap: PixelMap | undefined = undefined;
12.
13.  loadImageWithUrl(url: string) {
14.  let responseCode = http.ResponseCode;
15.  let OutData: http.HttpResponse;
16.  let imagePackerApi = image.createImagePacker();
17.  let packOpts: image.PackingOption = { format: 'image/jpeg', quality: 98 };
18.  // 确保网络正常
19.  http.createHttp().request(url, {
20.  method: http.RequestMethod.GET,
21.  connectTimeout: 60000,
22.  readTimeout: 60000
23.  },
24.  async (error: BusinessError, data: http.HttpResponse) => {
25.  if (error) {
26.  console.error(`http reqeust failed with. Code: ${error.code}, message: ${error.message}`);
27.  } else {
28.  OutData = data;
29.  let code: http.ResponseCode | number = OutData.responseCode;
30.  if (responseCode.OK === code) {
31.  let imageData: ArrayBuffer = OutData.result as ArrayBuffer;
32.  let imageSource: image.ImageSource = image.createImageSource(imageData);
33.
34.  class tmp {
35.  height: number = 100
36.  width: number = 100
37.  };
38.
39.  let options: Record<string, number | boolean | tmp> = {
40.  'alphaType': 0, // 透明度
41.  'editable': false, // 是否可编辑
42.  'pixelFormat': 3, // 像素格式
43.  'scaleMode': 1, // 缩略值
44.  'size': { height: 100, width: 100 }
45.  }; // 创建图片大小
46.  imageSource.createPixelMap(options).then((pixelMap: PixelMap) => {
47.  this.pixelMap = pixelMap;
48.  this.pixelMap.getImageInfo().then((info: image.ImageInfo) => {
49.  console.info('info.width = ' + info.size.width);
50.  }).catch((err: BusinessError) => {
51.  console.error('Failed ' + err);
52.  })
53.  imagePackerApi.packing(pixelMap, packOpts).then(async (buffer: ArrayBuffer) => {
54.  try {
55.  const context = getContext(this);
56.  let helper = photoAccessHelper.getPhotoAccessHelper(context);
57.  let uri = await helper.createAsset(photoAccessHelper.PhotoType.IMAGE, 'png');
58.  let file = await fs.open(uri, fs.OpenMode.READ_WRITE | fs.OpenMode.CREATE);
59.  // 写入文件
60.  await fs.write(file.fd, buffer);
61.  promptAction.showToast({ message: '已保存至相册！' });
62.  // 关闭文件
63.  await fs.close(file.fd);
64.  } catch (error) {
65.  console.error('error is ' + JSON.stringify(error));
66.  }
67.  }).catch((error: BusinessError) => {
68.  console.error('Failed to pack the image. And the error is: ' + error);
69.  })
70.  pixelMap.release();
71.  })
72.  }
73.  }
74.  }
75.  )
76.  }
77.
78.  build() {
79.  Row() {
80.  Column({ space: 10 }) {
81.  Image('https://agc-storage-drcn.platform.dbankcloud.cn/v0/test-rqcjj/test.png')
82.  .width('80%')
83.
84.  SaveButton().onClick(async (event: ClickEvent, result: SaveButtonOnClickResult) => {
85.  if (result === SaveButtonOnClickResult.SUCCESS) {
86.  this.loadImageWithUrl('https://agc-storage-drcn.platform.dbankcloud.cn/v0/test-rqcjj/test.png');
87.  } else {
88.  promptAction.showToast({ message: '设置权限失败！' });
89.  }
90.  })
91.  }
92.  .width('100%')
93.  }
94.  .height('100%')
95.  .backgroundColor(0xF1F3F5)
96.  }
97. }

```
